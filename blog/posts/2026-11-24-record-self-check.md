## 점검은 학기 말에 하는 게 좋아요

학년이 끝나 갈 즈음이면 그 해 학생부가 거의 다 채워져요. 이때 한 번 돌아보면 **이번 학년에 뭐가 남았고, 다음 학년에 뭘 채울지**가 보여요.

아래 18개 문장을 읽고, **지금 당장 "응, 나 그래"라고 말할 수 있는 것만** 체크해 보세요. 애매하면 비워 두는 게 더 정확해요. 비어 있는 칸은 점수가 아니라 다음에 할 일 목록이에요.

## 결과는 이렇게 읽어요

| 영역별 체크 | 지금 상태 | 다음에 할 일 |
|---|---|---|
| 5~6개 | 길이 잘 보여요 | 다음 학기 계획을 미리 적어 두기 |
| 3~4개 | 빈칸이 군데군데 보여요 | 비어 있는 문장 하나만 골라 이번 달에 채우기 |
| 0~2개 | 방향부터 다시 잡을 때예요 | A 영역(방향 · 과목)부터 차근차근 |

> 한꺼번에 다 채우려고 하면 기록이 오히려 흩어져요.
> 가장 비어 있는 영역 하나부터 시작하세요.

<script>
(function () {
  var root = document.getElementById('selfCheck'); if (!root) return;
  var secs = root.querySelectorAll('.sc-sec'), total = root.querySelector('.sc-total');
  var NAME = { dir: 'A. 방향 · 과목', 'class': 'B. 수업 속 기록', flow: 'C. 학생부 전체 흐름' };
  function level(n) { return n >= 5 ? ['good', '길이 잘 보여요'] : n >= 3 ? ['mid', '빈칸이 군데군데 보여요'] : ['low', '방향부터 다시 잡을 때예요']; }
  function update() {
    var any = false, worst = null;
    secs.forEach(function (s) {
      var n = s.querySelectorAll('input:checked').length, all = s.querySelectorAll('input').length, r = s.querySelector('.sc-res');
      if (n) any = true;
      var lv = level(n);
      r.className = 'sc-res ' + lv[0]; r.textContent = n + ' / ' + all + ' · ' + lv[1];
      if (!worst || n < worst.n) worst = { n: n, key: s.dataset.sec };
    });
    if (!any) { total.innerHTML = '<b>체크해 보세요</b><span>체크하면 영역별 결과가 바로 나와요. 체크한 내용은 어디에도 저장되지 않아요.</span>'; return; }
    total.innerHTML = '<b>가장 비어 있는 영역: ' + NAME[worst.key] + '</b><span>이번 달에는 이 영역의 빈 문장 하나만 골라 채워 보세요.</span>';
  }
  root.addEventListener('change', function () { update(); try { window.hfTrack && hfTrack('blog_selfcheck', { post: 'record-self-check' }); } catch (e) {} });
})();
</script>
